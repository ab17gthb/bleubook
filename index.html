How to set up the Google Sheets Database (Click to expand)

Because exams with images are too large for standard links, this app saves your exams directly to a hidden tab in your Google Sheet, and uses other tabs to collect student answers.

  1. Create a new Google Sheet and click Extensions > Apps Script.
  2. Delete any code there, and paste the Database code block below.
  3. Click the Save icon.
  4. Click Deploy > New deployment (Select type: Web app).
  5. Set "Execute as" to Me, and "Who has access" to Anyone.
  6. Click Deploy, authorize the permissions, and copy the Web App URL and paste it into the box above!
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  
  if (data.action === "saveExam") {
    var dbSheet = ss.getSheetByName("Exam_Database");
    if (!dbSheet) {
      dbSheet = ss.insertSheet("Exam_Database");
      dbSheet.appendRow(["Exam ID", "Exam Data"]);
    }
    var examId = "EXAM-" + Math.floor(Math.random() * 1000000000);
    dbSheet.appendRow([examId, JSON.stringify(data.examData)]);
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "examId": examId}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (data.action === "getExam") {
    var dbSheet = ss.getSheetByName("Exam_Database");
    var records = dbSheet.getDataRange().getValues();
    for (var i = 1; i < records.length; i++) {
      if (records[i][0] === data.examId) {
        return ContentService.createTextOutput(records[i][1]).setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({"error": "Exam not found"})).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (data.action === "submitAnswers") {
    var examTitle = data.examTitle || "Untitled Exam";
    var sheet = ss.getSheetByName(examTitle);
    
    if (!sheet) {
      sheet = ss.insertSheet(examTitle);
      var headers = ["Timestamp", "Student Name", "Exam Title"];
      for (var i = 0; i < data.responses.length; i++) {
        if (data.prompts && data.prompts[i]) {
          headers.push(data.prompts[i]); 
        } else {
          headers.push("Response " + (i + 1));
        }
      }
      sheet.appendRow(headers);
      sheet.setFrozenRows(1);
      sheet.getRange("1:1").setFontWeight("bold");
    }
    
    var row = [new Date(), data.studentName, data.examTitle];
    for (var i = 0; i < data.responses.length; i++) {
      row.push(data.responses[i]);
    }
    sheet.appendRow(row);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
